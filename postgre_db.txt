-- ---------------------------------------------------------------------
-- Shared trigger function: emulates MySQL's ON UPDATE CURRENT_TIMESTAMP
-- ---------------------------------------------------------------------
CREATE OR REPLACE FUNCTION set_updated_at()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
    NEW.updated_at = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$;


--  USERS (MUST BE FIRST – referenced by all other tables)
CREATE TABLE users (
    user_id SERIAL PRIMARY KEY,

    -- Postgres string comparison is case-sensitive by default, so the
    -- MySQL COLLATE utf8mb4_bin on username is implicit here.
    username VARCHAR(50) NOT NULL,
    role VARCHAR(20) NOT NULL DEFAULT 'Grounds Staff'
        CHECK (role IN ('Administrator', 'Office Staff', 'Grounds Staff')),
    status VARCHAR(20) NOT NULL DEFAULT 'Unverified'
        CHECK (status IN ('Verified', 'Unverified')),
    email VARCHAR(255) NOT NULL,
    phone_number VARCHAR(20) NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    name VARCHAR(255) NOT NULL,
    session_token TEXT NULL,

    -- Audit columns
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP NULL,
    created_by INT NULL,
    updated_by INT NULL,

    -- Shadow columns: unique only among ACTIVE rows.
    -- PG allows many NULLs in a UNIQUE index, so soft-deleted rows
    -- don't collide with each other.
    active_username VARCHAR(50)
        GENERATED ALWAYS AS (CASE WHEN deleted_at IS NULL THEN username ELSE NULL END) STORED,
    active_email VARCHAR(255)
        GENERATED ALWAYS AS (CASE WHEN deleted_at IS NULL THEN email ELSE NULL END) STORED,
    active_phone VARCHAR(20)
        GENERATED ALWAYS AS (CASE WHEN deleted_at IS NULL THEN phone_number ELSE NULL END) STORED,

    CONSTRAINT uk_active_username UNIQUE (active_username),
    CONSTRAINT uk_active_email    UNIQUE (active_email),
    CONSTRAINT uk_active_phone    UNIQUE (active_phone),

    CONSTRAINT fk_users_created_by FOREIGN KEY (created_by)
        REFERENCES users(user_id) ON DELETE SET NULL,
    CONSTRAINT fk_users_updated_by FOREIGN KEY (updated_by)
        REFERENCES users(user_id) ON DELETE SET NULL
);

CREATE TRIGGER trg_users_updated_at
BEFORE UPDATE ON users
FOR EACH ROW EXECUTE FUNCTION set_updated_at();

--  BLOCKS (sections / areas)
CREATE TABLE blocks (
    block_id SERIAL PRIMARY KEY,
    block_name VARCHAR(100) NOT NULL,
    block_type VARCHAR(30) NOT NULL
        CHECK (block_type IN (
            'Niche', 'Bone Chamber', 'Lawn/Grounds',
            'Unmapped Area', 'Private', 'Mausoleum',
            'Mass Grave', 'Cluster', 'Block'
        )),
    image_link TEXT,
    coordinates JSONB,
    "rows" INT NOT NULL DEFAULT 0,
    "cols" INT NOT NULL DEFAULT 0,
    rotation DECIMAL(10,6) DEFAULT 0.000000,
    remarks TEXT,

    -- Audit columns
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP NULL,
    created_by INT NULL,
    updated_by INT NULL,

    active_block_name VARCHAR(100)
        GENERATED ALWAYS AS (CASE WHEN deleted_at IS NULL THEN block_name ELSE NULL END) STORED,
    CONSTRAINT uk_active_block_name UNIQUE (active_block_name),

    CONSTRAINT fk_blocks_created_by FOREIGN KEY (created_by)
        REFERENCES users(user_id) ON DELETE SET NULL,
    CONSTRAINT fk_blocks_updated_by FOREIGN KEY (updated_by)
        REFERENCES users(user_id) ON DELETE SET NULL
);

CREATE TRIGGER trg_blocks_updated_at
BEFORE UPDATE ON blocks
FOR EACH ROW EXECUTE FUNCTION set_updated_at();

    created_by INT NULL,
    updated_by INT NULL,

    active_grave_code VARCHAR(50)
        GENERATED ALWAYS AS (CASE WHEN deleted_at IS NULL THEN grave_code ELSE NULL END) STORED,

--  INTERMENTS (PRESENT — current state)
CREATE TABLE interments (
    interment_id SERIAL PRIMARY KEY,
    control_number VARCHAR(100) NOT NULL,

    deceased_name VARCHAR(255) NOT NULL,
    last_known_address TEXT,
    death_certificate VARCHAR(100),
    deceased_date_of_birth DATE,
    deceased_date_of_death DATE,
    deceased_sex VARCHAR(10) DEFAULT 'Unknown'
        CHECK (deceased_sex IN ('Male', 'Female', 'Unknown')),

    current_grave_id INT NULL,

    contact_person_name VARCHAR(255),
    contact_person_phone_number VARCHAR(50),
    contact_person_email VARCHAR(150),
    contact_person_address_barangay TEXT,
    contact_person_address TEXT,

    assistance_type VARCHAR(50) NOT NULL DEFAULT 'Burial'
        CHECK (assistance_type IN ('Burial', 'Transfer the remains of the late', 'Other')),

    burial_permit_number VARCHAR(100),
    burial_permit_date DATE,
    transfer_permit_number VARCHAR(100),
    transfer_permit_issued_by VARCHAR(150),
    transfer_permit_date DATE,
    exhumation_permit_number VARCHAR(100),
    exhumation_permit_date DATE,

    date_buried DATE,
    date_exhumed DATE,
    burial_clearance_date DATE,
    lease_expiration_date DATE,

    status VARCHAR(20) NOT NULL DEFAULT 'Pending'
        CHECK (status IN ('Pending', 'Active', 'Inactive')),

    remarks TEXT,

    -- Audit columns
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP NULL,
    created_by INT NULL,
    updated_by INT NULL,

    active_control_number VARCHAR(100)
        GENERATED ALWAYS AS (CASE WHEN deleted_at IS NULL THEN control_number ELSE NULL END) STORED,
    CONSTRAINT uk_active_control_number UNIQUE (active_control_number),

    CONSTRAINT fk_interments_grave FOREIGN KEY (current_grave_id)
        REFERENCES graves(grave_id) ON DELETE RESTRICT,
    CONSTRAINT fk_interments_created_by FOREIGN KEY (created_by)
        REFERENCES users(user_id) ON DELETE SET NULL,
    CONSTRAINT fk_interments_updated_by FOREIGN KEY (updated_by)
        REFERENCES users(user_id) ON DELETE SET NULL,

    -- If status is Pending or Inactive, current_grave_id must be NULL.
    CONSTRAINT chk_status_grave CHECK (
        status = 'Active' OR current_grave_id IS NULL
    )
);

CREATE TRIGGER trg_interments_updated_at
BEFORE UPDATE ON interments
FOR EACH ROW EXECUTE FUNCTION set_updated_at();


--  RESERVATIONS (FUTURE — planned data)
CREATE TABLE reservations (
    reservation_id SERIAL PRIMARY KEY,

    pending_interment_id INT NOT NULL,
    target_grave_id INT NOT NULL,
    old_interment_id INT NULL,

    old_new_grave_id INT NULL,
    old_new_status VARCHAR(20) NULL
        CHECK (old_new_status IS NULL OR old_new_status IN ('Active', 'Inactive')),
    old_new_assistance_type VARCHAR(50) NULL
        CHECK (old_new_assistance_type IS NULL OR old_new_assistance_type IN
            ('Burial', 'Transfer the remains of the late', 'Other')),
    old_new_remarks TEXT NULL,

    exhumation_permit_number VARCHAR(100) NULL,
    transfer_permit_number VARCHAR(100) NULL,

    reservation_notes TEXT NULL,

    reservation_state VARCHAR(20) NOT NULL DEFAULT 'Active'
        CHECK (reservation_state IN ('Active', 'Executed', 'Cancelled')),
    actioned_at TIMESTAMP NULL,
    actioned_by INT NULL,

    old_edited_fields JSONB NULL,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    created_by INT NULL,
    updated_by INT NULL,

    -- Shadow columns: uniqueness only among Active reservations.
    active_pending_interment_id INT
        GENERATED ALWAYS AS (CASE WHEN reservation_state = 'Active' THEN pending_interment_id ELSE NULL END) STORED,
    active_target_grave_id INT
        GENERATED ALWAYS AS (CASE WHEN reservation_state = 'Active' THEN target_grave_id ELSE NULL END) STORED,
    active_old_interment_id INT
        GENERATED ALWAYS AS (CASE WHEN reservation_state = 'Active' THEN old_interment_id ELSE NULL END) STORED,

    -- If O is not set, nothing about O is set. If O is set, its new
    -- status must be decided.
    CONSTRAINT chk_old_pairing CHECK (
        (old_interment_id IS NULL
            AND old_new_grave_id IS NULL
            AND old_new_status IS NULL
            AND old_new_assistance_type IS NULL)
        OR
        (old_interment_id IS NOT NULL AND old_new_status IS NOT NULL)
    ),

    -- An Inactive O has no grave.
    CONSTRAINT chk_inactive_has_no_grave CHECK (
        NOT (old_new_status = 'Inactive' AND old_new_grave_id IS NOT NULL)
    ),

    -- O's destination must differ from the target grave.
    CONSTRAINT chk_old_new_grave_not_target CHECK (
        old_new_grave_id IS NULL OR old_new_grave_id <> target_grave_id
    ),

    CONSTRAINT uk_active_pending_interment UNIQUE (active_pending_interment_id),
    CONSTRAINT uk_active_target_grave      UNIQUE (active_target_grave_id),
    CONSTRAINT uk_active_old_interment     UNIQUE (active_old_interment_id),

    CONSTRAINT fk_res_pending FOREIGN KEY (pending_interment_id)
        REFERENCES interments(interment_id) ON DELETE RESTRICT,
    CONSTRAINT fk_res_target FOREIGN KEY (target_grave_id)
        REFERENCES graves(grave_id) ON DELETE RESTRICT,
    CONSTRAINT fk_res_old FOREIGN KEY (old_interment_id)
        REFERENCES interments(interment_id) ON DELETE RESTRICT,
    CONSTRAINT fk_res_old_new_grave FOREIGN KEY (old_new_grave_id)
        REFERENCES graves(grave_id) ON DELETE RESTRICT,
    CONSTRAINT fk_res_actioned_by FOREIGN KEY (actioned_by)
        REFERENCES users(user_id) ON DELETE SET NULL,
    CONSTRAINT fk_res_created_by FOREIGN KEY (created_by)
        REFERENCES users(user_id) ON DELETE SET NULL,
    CONSTRAINT fk_res_updated_by FOREIGN KEY (updated_by)
        REFERENCES users(user_id) ON DELETE SET NULL
);

CREATE TRIGGER trg_reservations_updated_at
BEFORE UPDATE ON reservations
FOR EACH ROW EXECUTE FUNCTION set_updated_at();

--  SMS_NOTIFICATIONS (append-only log of every automated or manual SMS)
CREATE TABLE sms_notifications (
    notification_id SERIAL PRIMARY KEY,

    interment_id INT NOT NULL,

    notification_type VARCHAR(30) NOT NULL DEFAULT 'Lease Expiring'
        CHECK (notification_type IN (
            'Lease Expiring',
            'Burial Confirmation',
            'Payment Reminder',
            'Other'
        )),

    lease_expiration_date_at_send DATE NULL,
    days_before_expiry INT NULL,

    recipient_name VARCHAR(255) NULL,
    recipient_phone_number VARCHAR(50) NOT NULL,
    message_body TEXT NOT NULL,

    status VARCHAR(20) NOT NULL DEFAULT 'Pending'
        CHECK (status IN ('Pending', 'Sent', 'Failed', 'Skipped')),
    sent_at TIMESTAMP NULL,
    provider VARCHAR(50) NULL DEFAULT 'TextBee',
    provider_message_id VARCHAR(255) NULL,
    error_message TEXT NULL,
    attempt_count INT NOT NULL DEFAULT 0,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    created_by INT NULL,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    -- Dedup key: only exists when status='Sent'.
    -- MySQL used YEAR()/MONTH()/DAY()/LPAD(); PG uses EXTRACT + || because
    -- generated columns must be immutable (a bare ::text cast of DATE is
    -- not, since it honours the DateStyle session setting).
    active_sent_key VARCHAR(255)
        GENERATED ALWAYS AS (
            CASE WHEN status = 'Sent' THEN
                interment_id::text || ':' || notification_type || ':' ||
                COALESCE(
                    LPAD(EXTRACT(YEAR  FROM lease_expiration_date_at_send)::int::text, 4, '0') || '-' ||
                    LPAD(EXTRACT(MONTH FROM lease_expiration_date_at_send)::int::text, 2, '0') || '-' ||
                    LPAD(EXTRACT(DAY   FROM lease_expiration_date_at_send)::int::text, 2, '0'),
                    'NULL'
                ) || ':' ||
                COALESCE(days_before_expiry::text, '-1')
            ELSE NULL END
        ) STORED,
    CONSTRAINT uk_active_sent_key UNIQUE (active_sent_key),

    CONSTRAINT fk_sms_interment FOREIGN KEY (interment_id)
        REFERENCES interments(interment_id) ON DELETE RESTRICT,
    CONSTRAINT fk_sms_created_by FOREIGN KEY (created_by)
        REFERENCES users(user_id) ON DELETE SET NULL
);

CREATE TRIGGER trg_sms_notifications_updated_at
BEFORE UPDATE ON sms_notifications
FOR EACH ROW EXECUTE FUNCTION set_updated_at();

COMMENT ON TABLE blocks            IS 'Cemetery sections/areas';
COMMENT ON TABLE graves            IS 'Individual burial slots inside a block';
COMMENT ON TABLE interments        IS 'PRESENT. One row per deceased. Current state.';
COMMENT ON TABLE reservations      IS 'FUTURE. Proposal to move O and place N. Frozen after creation.';
COMMENT ON TABLE placement_history IS 'PAST. Append-only placement history. No edits logged.';
COMMENT ON TABLE settings          IS 'System configuration key-value store';
COMMENT ON TABLE payments          IS 'Payment records with dual-approval workflow';
COMMENT ON TABLE sms_notifications IS 'PAST. Append-only log of every SMS attempt (lease reminders, etc).';
